import axios from 'axios';
import * as bitcoin from 'bitcoinjs-lib';
import { ECPairInterface } from 'ecpair';
import * as ecc from 'tiny-secp256k1';

// Initialize bitcoinjs-lib with tiny-secp256k1
bitcoin.initEccLib(ecc);

class BitcoinTestnetService {
    private baseUrl: string;
    private network: bitcoin.Network;

    constructor() {
        this.baseUrl = 'https://blockstream.info/testnet/api';
        this.network = bitcoin.networks.testnet;
    }

    /**
     * Get the balance of a Bitcoin address.
     */
    async getBalance(address: string): Promise<number> {
        const url = `${this.baseUrl}/address/${address}/utxo`;
        const response = await axios.get(url);
        const utxos = response.data;

        let balance = 0;
        utxos.forEach((utxo: any) => {
            balance += utxo.value;
        });

        return balance;
    }

    /**
     * Get transactions for a Bitcoin address.
     */
    async getTransactions(address: string): Promise<any[]> {
        const url = `${this.baseUrl}/address/${address}/txs`;
        const response = await axios.get(url);
        return response.data;
    }

    /**
     * Create a raw transaction.
     */
    createTransaction(
        utxos: any[],
        toAddress: string,
        amount: number,
        changeAddress: string,
        feeRate: number = 10 // satoshis per byte
    ): bitcoin.Psbt {
        const psbt = new bitcoin.Psbt({ network: this.network });

        let totalInput = 0;
        utxos.forEach((utxo) => {
            totalInput += utxo.value;
            psbt.addInput({
                hash: utxo.txid,
                index: utxo.vout,
                witnessUtxo: {
                    script: bitcoin.address.toOutputScript(utxo.address, this.network),
                    value: utxo.value,
                },
            });
        });

        const totalOutput = amount + feeRate * 100; // Estimate fee
        if (totalInput < totalOutput) {
            throw new Error('Insufficient funds');
        }

        psbt.addOutput({
            address: toAddress,
            value: amount,
        });

        // Add change output
        if (totalInput > totalOutput) {
            psbt.addOutput({
                address: changeAddress,
                value: totalInput - totalOutput,
            });
        }

        return psbt;
    }

    /**
     * Sign a transaction with a private key.
     */
    signTransaction(psbt: bitcoin.Psbt, keyPair: ECPairInterface): bitcoin.Psbt {
        psbt.data.inputs.forEach((_, index) => {
            psbt.signInput(index, keyPair);
        });
        return psbt;
    }

    /**
     * Broadcast a signed transaction to the Bitcoin network.
     */
    async broadcastTransaction(hex: string): Promise<string> {
        const url = `${this.baseUrl}/tx`;
        const response = await axios.post(url, hex, {
            headers: { 'Content-Type': 'text/plain' },
        });
        return response.data;
    }

    /**
     * Send a transaction (combines create, sign, and broadcast).
     */
    async sendTransaction(
        utxos: any[],
        toAddress: string,
        amount: number,
        changeAddress: string,
        keyPair: ECPairInterface
    ): Promise<string> {
        const psbt = this.createTransaction(utxos, toAddress, amount, changeAddress);
        const signedPsbt = this.signTransaction(psbt, keyPair);
        const finalizedPsbt = signedPsbt.finalizeAllInputs();
        const txHex = finalizedPsbt.extractTransaction().toHex();
        return this.broadcastTransaction(txHex);
    }
}

// Usage Example
(async () => {
    const bitcoinService = new BitcoinTestnetService();

    // Example: Get balance

        },

Explanation of Methods:
getBalance:

Fetches the UTXOs (Unspent Transaction Outputs) for a given address and calculates the total balance.

getTransactions:

Retrieves the transaction history for a given address.

createTransaction:

Creates a raw transaction (PSBT - Partially Signed Bitcoin Transaction) using UTXOs, a recipient address, and a change address.


signTransaction:

Signs the transaction using a private key (ECPair).

broadcastTransaction:

Sends the signed transaction to the Bitcoin network.

sendTransaction:

Combines createTransaction, signTransaction, and broadcastTransaction into a single method for convenience.

Notes:
Dependencies: Ensure you have bitcoinjs-lib, tiny-secp256k1, and axios installed.

Key Pair: Replace the keyPair with your actual private key or use a library like bip39 to derive it from a mnemonic.

Error Handling: Add proper error handling for production use.

Fee Calculation: The fee calculation is simplified. You can improve it by dynamically calculating the transaction size.





import * as bitcoin from 'bitcoinjs-lib';
import { BIP32Interface } from 'bip32';
import * as ecc from 'tiny-secp256k1';
import StrategySelector from './walletGenerationStrategies/strategySelector.js';
import { httpClient } from '../utils/httpClient.js';
export class BitcoinWallet {
    private network: bitcoin.Network;
    private root: BIP32Interface;
    constructor(strategySelector: StrategySelector, network: bitcoin.Network) {
        this.network = network;
        const strategy = strategySelector.selectStrategy();
        this.root = strategy.generateRoot();
    }

    getNetwork(): bitcoin.Network {
        return this.network;
    }
    getRoot(): BIP32Interface {
        return this.root;
    }

    async createTransaction(
        recipientAddress: string,
        amount: number,
        fee: number,
        utxos: any[]
    ): Promise<string> { // Return the transaction ID after broadcasting
        const psbt = new bitcoin.Psbt({ network: this.network });

        let totalInput = 0;
        console.log(utxos);
        // Add inputs (UTXOs)
        for (const utxo of utxos) {
            const input: any = {
                hash: utxo.txid,
                index: utxo.vout,
            };

            // Check if the script is SegWit (P2WPKH or P2WSH)
            if (utxo.scriptPubKey.startsWith('0014') || utxo.scriptPubKey.startsWith('0020')) {
                // SegWit script: Use witnessUtxo
                input.witnessUtxo = {
                    script: Buffer.from(utxo.scriptPubKey, 'hex'),
                    value: utxo.value,
                };
            } else {
                // Non-SegWit script: Use nonWitnessUtxo
                const txHex = await this.fetchTransactionHex(utxo.txid); // Fetch the full transaction hex
                input.nonWitnessUtxo = Buffer.from(txHex, 'hex');
            }

            psbt.addInput(input);
            totalInput += utxo.value;
        }

        const remainingValue = totalInput - Number(amount) - fee;


        if (remainingValue < 0) {
            throw new Error('Insufficient funds to cover the amount and fee.');
        }

        // Validate recipient address
        try {
            bitcoin.address.toOutputScript(recipientAddress, this.network);
        } catch (error) {
            throw new Error(`Invalid recipient address: ${recipientAddress}`);
        }

        try {
            psbt.addOutput({
                address: recipientAddress,
                value: Number(amount),
            });
            console.log('Recipient output added successfully.');
        } catch (error) {
            console.error('Error adding recipient output:', error);
            throw error;
        }

        if (remainingValue > 0) {
            try {
                psbt.addOutput({
                    address: this.deriveAdress(utxos.length),
                    value: remainingValue,
                });
                console.log('Change output added successfully.');
            } catch (error) {
                console.error('Error adding change output:', error);
                throw error;
            }
        }

        for (let i = 0; i < utxos.length; i++) {
            let path: string;
          
            // Determine the derivation path based on the UTXO address
            const utxoAddress = utxos[i].address;
            const isMainAddress = utxoAddress === this.deriveMainAddress(0); // Check if the UTXO belongs to the main address
          
            if (isMainAddress) {
              // Use the main address derivation path (m/44'/1'/0'/0/index)
              path = `m/44'/1'/0'/0/${i}`;
            } else {
              // Use the change address derivation path (m/44'/1'/0'/1/index)
              path = `m/44'/1'/0'/1/${i}`;
            }
          
            // Derive the child node
            const child = this.root.derivePath(path);
            
          
            // Get the address from the derived public key
            const derivedAddress = bitcoin.payments.p2wpkh({
              pubkey: Buffer.from(child.publicKey), // Convert Uint8Array to Buffer
              network: this.network,
            }).address;
            console.log('Derived Address:', derivedAddress);
            console.log('UTXO Address:', utxos[i].address);
          
            if (!child.privateKey) {
              throw new Error('Private key not available for signing');
            }
          
            // Create a signer object
            const signer = {
              publicKey: Buffer.from(child.publicKey), // Convert Uint8Array to Buffer
              sign: (hash: Buffer) => {
                const signature = ecc.sign(hash, child.privateKey!);
                return Buffer.from(signature); // Convert Uint8Array to Buffer
              },
            };
          
            // Sign the input
            psbt.signInput(i, signer);
            console.log(`Input ${i} signed successfully.`);
          }
          
        
        // Finalize and extract the transaction
        psbt.finalizeAllInputs();
        const tx = psbt.extractTransaction();

        const rawTx = tx.toHex();
        const txId = await this.broadcastTransaction(rawTx);

        return txId;
    }

    deriveAdress(index: number) {
        if (index >= 1)
            index = index - 1;
        const changeAddress = bitcoin.payments.p2wpkh({
            pubkey: Buffer.from(this.root.derive(1).derive(index).publicKey), // Derive change address (path m/44'/1'/0'/1/index)
            network: this.network,
        }).address!;
        return changeAddress;
    }

     deriveMainAddress(index:number): string {
        const mainAddress = bitcoin.payments.p2wpkh({
            pubkey: Buffer.from(this.root.derive(0).derive(index).publicKey), // Derive main address (path m/44'/1'/0'/0/index)
            network: this.network,
        }).address!;
        return mainAddress;
       
    }
    async fetchTransactionHex(txid: string): Promise<string> {
        const url = `/tx/${txid}/hex`;
        const response = await httpClient.get(url);
        return response.data;
    }

    async broadcastTransaction(hex: string): Promise<string> {
        try {
            const url = `/tx`;
            const response = await httpClient.post(url, hex, {
                headers: { 'Content-Type': 'text/plain' },
            });
            return response.data;
        } catch (error) {
            if (error instanceof Error) {
                console.error('Error broadcasting transaction:', error.message);
            } else {
                console.error('An unknown error occurred:', error);
            }
            throw error; // Re-throw the error to propagate it
        }
    }
}
